#include <SPI.h>
#include <MFRC522.h>
#include <Wire.h>
#include <LiquidCrystal_I2C.h>

#define SS_PIN 10
#define RST_PIN 9
#define GREEN 8
#define RED 7

MFRC522 rfid(SS_PIN, RST_PIN);
LiquidCrystal_I2C lcd(0x27, 16, 2);

String cardIDs[] = {"A1B2C3D4", "11223344"};
String names[]   = {"Rahul", "Khushi"};
bool present[]   = {false, false};
const int TOTAL = 2;

void showHome() {
  lcd.clear();
  lcd.print("RFID Attendance");
  lcd.setCursor(0, 1);
  lcd.print("Scan your card");
}

void setup() {
  Serial.begin(9600);
  SPI.begin();
  rfid.PCD_Init();
  lcd.init();
  lcd.backlight();
  pinMode(GREEN, OUTPUT);
  pinMode(RED, OUTPUT);
  showHome();
}

void loop() {
  if (!rfid.PICC_IsNewCardPresent() || !rfid.PICC_ReadCardSerial()) return;

  String uid = "";
  for (byte i = 0; i < rfid.uid.size; i++) {
    if (rfid.uid.uidByte[i] < 0x10) uid += "0";
    uid += String(rfid.uid.uidByte[i], HEX);
  }
  uid.toUpperCase();
  Serial.println("UID: " + uid);

  int found = -1;
  for (int i = 0; i < TOTAL; i++) {
    if (cardIDs[i] == uid) found = i;
  }

  lcd.clear();
  if (found == -1) {
    lcd.print("Unknown Card!");
    lcd.setCursor(0, 1);
    lcd.print(uid);
    digitalWrite(RED, HIGH);
    delay(2000);
    digitalWrite(RED, LOW);
  } else if (present[found]) {
    lcd.print(names[found]);
    lcd.setCursor(0, 1);
    lcd.print("Already Marked");
    delay(1500);
  } else {
    present[found] = true;
    lcd.print("Welcome " + names[found]);
    lcd.setCursor(0, 1);
    lcd.print("Present!");
    digitalWrite(GREEN, HIGH);
    delay(1500);
    digitalWrite(GREEN, LOW);
  }

  rfid.PICC_HaltA();
  showHome();
}
